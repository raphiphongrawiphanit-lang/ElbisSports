    <script>
        // Global App State - Each team: { name: string, affiliation: string }
        let teams = [];
        let matches = [];
        let rrMatches = [];

        // Parse text line into { name, affiliation }
        function parseParticipantText(str) {
            str = str.trim();
            if (!str) return null;

            // 1. Check bracket format: Name (Team)
            const bracketMatch = str.match(/^(.*?)\s*[\(\（](.*?)[\)\）]\s*$/);
            if (bracketMatch && bracketMatch[1].trim()) {
                return { name: bracketMatch[1].trim(), affiliation: bracketMatch[2].trim() };
            }

            // 2. Check Tab separator (Excel / Google Sheets paste)
            if (str.includes('\t')) {
                const parts = str.split('\t');
                return { name: parts[0].trim(), affiliation: parts.slice(1).join(' ').trim() };
            }

            // 3. Check explicit delimiters: ' - ', '-', ',', '|'
            const delimiters = [' - ', '-', ',', '|'];
            for (let d of delimiters) {
                if (str.includes(d)) {
                    const parts = str.split(d);
                    const name = parts[0].trim();
                    const affiliation = parts.slice(1).join(d).trim();
                    if (name) {
                        return { name, affiliation };
                    }
                }
            }

            // 4. Check multiple spaces (2 or more spaces)
            if (/\s{2,}/.test(str)) {
                const parts = str.split(/\s{2,}/);
                return { name: parts[0].trim(), affiliation: parts.slice(1).join(' ').trim() };
            }

            // 5. Fallback for single space separation: "Name Team"
            if (str.includes(' ')) {
                const firstSpaceIdx = str.indexOf(' ');
                const name = str.substring(0, firstSpaceIdx).trim();
                const affiliation = str.substring(firstSpaceIdx + 1).trim();
                if (name && affiliation) {
                    return { name, affiliation };
                }
            }

            return { name: str, affiliation: '' };
        }

        // Safe Initialization Handler
        function initApp() {
            teams = [];
            for (let i = 1; i <= 8; i++) {
                teams.push({
                    name: `ผู้แข่งขัน ${i}`,
                    affiliation: `สังกัด ${String.fromCharCode(64 + i)}`
                });
            }
            updateTeamsUI();
            generateBracket();
        }

        if (document.readyState === 'complete' || document.readyState === 'interactive') {
            setTimeout(initApp, 50);
        } else {
            document.addEventListener('DOMContentLoaded', initApp);
        }

        // Add Single Team
        function addTeam() {
            const nameInput = document.getElementById('new-team-input');
            const affInput = document.getElementById('new-team-affiliation');
            
            const rawName = nameInput.value.trim();
            const rawAff = affInput.value.trim();

            if (!rawName) return;

            if (!rawAff) {
                const parsed = parseParticipantText(rawName);
                if (parsed) {
                    teams.push(parsed);
                } else {
                    teams.push({ name: rawName, affiliation: '' });
                }
            } else {
                teams.push({
                    name: rawName,
                    affiliation: rawAff
                });
            }

            nameInput.value = '';
            affInput.value = '';
            nameInput.focus();
            updateTeamsUI();
            generateBracket();
        }

        // Add Bulk Teams
        function addBulkTeams() {
            const textarea = document.getElementById('bulk-teams-input');
            const lines = textarea.value.split('\n');
            
            let addedCount = 0;
            lines.forEach(line => {
                const item = parseParticipantText(line);
                if (item) {
                    teams.push(item);
                    addedCount++;
                }
            });

            if (addedCount > 0) {
                textarea.value = '';
                updateTeamsUI();
                generateBracket();
            }
        }

        // Remove Team
        function removeTeam(index) {
            teams.splice(index, 1);
            updateTeamsUI();
            generateBracket();
        }

        // Clear All Teams
        function clearAllTeams() {
            teams = [];
            updateTeamsUI();
            generateBracket();
        }

        // Helper to format participant item object
        function getParticipantObj(item) {
            if (typeof item === 'string') {
                return parseParticipantText(item) || { name: item, affiliation: '' };
            }
            return item || { name: '', affiliation: '' };
        }

        // Helper to display label string
        function getDisplayLabel(item, num) {
            const p = getParticipantObj(item);
            if (!p.name) return '';
            const prefix = num ? `${num}. ` : '';
            return p.affiliation ? `${prefix}${p.name} (${p.affiliation})` : `${prefix}${p.name}`;
        }

        // Render Left Sidebar Teams List
        function updateTeamsUI() {
            const listEl = document.getElementById('teams-list');
            const countEl = document.getElementById('team-count');
            countEl.textContent = teams.length;

            if (teams.length === 0) {
                listEl.innerHTML = `<p class="text-xs text-slate-400 text-center py-5 border border-dashed border-slate-200 rounded-xl">ยังไม่มีรายชื่อผู้แข่งขัน</p>`;
                return;
            }

            listEl.innerHTML = teams.map((team, idx) => {
                const p = getParticipantObj(team);
                return `
                <div class="flex items-center justify-between p-2.5 bg-slate-50 border border-slate-200 rounded-xl text-xs hover:border-slate-300 transition">
                    <span class="font-medium text-slate-700 flex items-center gap-2 truncate">
                        <span class="w-5 h-5 rounded-full bg-red-100 text-red-600 text-[10px] font-bold flex items-center justify-center flex-shrink-0">
                            ${idx + 1}
                        </span>
                        <span class="truncate">
                            <strong class="text-slate-900 font-medium">${escapeHtml(p.name)}</strong>
                            ${p.affiliation ? `<span class="text-[10px] text-slate-500 bg-white border border-slate-200 px-1.5 py-0.5 rounded-md ml-1 truncate inline-block max-w-[110px]">${escapeHtml(p.affiliation)}</span>` : ''}
                        </span>
                    </span>
                    <button onclick="removeTeam(${idx})" class="text-slate-400 hover:text-red-600 font-medium px-1 flex-shrink-0 transition">
                        ลบ
                    </button>
                </div>
            `}).join('');
        }

        // Instant Random Shuffle & Bracket Regeneration
        function shuffleAndGenerate() {
            if (teams.length < 2) {
                alertModal("กรุณาเพิ่มรายชื่อผู้แข่งขันอย่างน้อย 2 ทีม");
                return;
            }
            shuffleArray(teams);
            updateTeamsUI();
            generateBracket();
        }

        // Fisher-Yates Shuffle Algorithm
        function shuffleArray(array) {
            for (let i = array.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [array[i], array[j]] = [array[j], array[i]];
            }
        }

        // Master Bracket Controller
        function generateBracket() {
            const format = document.getElementById('tournament-format').value;
            const titleInput = document.getElementById('tournament-name').value;

            document.getElementById('display-tournament-title').textContent = titleInput || "ผังการแข่งขัน";
            document.getElementById('stat-teams').textContent = teams.length;
            document.getElementById('champion-banner').classList.add('hidden');

            const bracketDisplay = document.getElementById('bracket-display');
            const emptyState = document.getElementById('empty-state');

            if (teams.length < 2) {
                bracketDisplay.innerHTML = '';
                emptyState.classList.remove('hidden');
                document.getElementById('stat-matches').textContent = '0';
                return;
            }

            emptyState.classList.add('hidden');

            if (format === 'knockout') {
                buildKnockoutBracket();
            } else if (format === 'roundrobin') {
                buildRoundRobinStandings();
            }
        }

        // Knockout (Single Elimination) - Sequential Top-to-Bottom Pairing with BYE
        function buildKnockoutBracket() {
            const participantList = [...teams];
            const numTeams = participantList.length;

            const totalRounds = Math.ceil(Math.log2(numTeams));
            const bracketSize = Math.pow(2, totalRounds);
            const numR1Matches = bracketSize / 2;

            matches = [];

            const numFullMatches = numTeams - numR1Matches;
            let r1Matches = [];
            let pIdx = 0;

            for (let i = 0; i < numR1Matches; i++) {
                let t1, t2, winner = null;

                if (i < numFullMatches) {
                    const p1 = getParticipantObj(participantList[pIdx]);
                    t1 = { name: p1.name, affiliation: p1.affiliation, num: pIdx + 1 };
                    pIdx++;

                    const p2 = getParticipantObj(participantList[pIdx]);
                    t2 = { name: p2.name, affiliation: p2.affiliation, num: pIdx + 1 };
                    pIdx++;
                } else {
                    const p1 = getParticipantObj(participantList[pIdx]);
                    t1 = { name: p1.name, affiliation: p1.affiliation, num: pIdx + 1 };
                    pIdx++;
                    t2 = { name: "BYE", affiliation: "", isBye: true };
                    winner = t1;
                }

                r1Matches.push({
                    id: `r1-m${i}`,
                    round: 1,
                    team1: t1,
                    team2: t2,
                    winner: winner
                });
            }
            matches.push(r1Matches);

            for (let r = 2; r <= totalRounds; r++) {
                const prevRound = matches[r - 2];
                let roundMatches = [];
                for (let i = 0; i < prevRound.length; i += 2) {
                    const prevM1 = prevRound[i];
                    const prevM2 = prevRound[i + 1];

                    let t1Obj = prevM1 && prevM1.winner ? prevM1.winner : { name: `ผู้ชนะ R${r-1}-M${i+1}`, affiliation: "", num: null };
                    let t2Obj = prevM2 && prevM2.winner ? prevM2.winner : { name: `ผู้ชนะ R${r-1}-M${i+2}`, affiliation: "", num: null };

                    roundMatches.push({
                        id: `r${r}-m${i/2}`,
                        round: r,
                        team1: t1Obj,
                        team2: t2Obj,
                        winner: null
                    });
                }
                matches.push(roundMatches);
            }

            for (let i = 0; i < r1Matches.length; i++) {
                if (r1Matches[i].winner) {
                    propagateWinner(0, r1Matches[i].id, r1Matches[i].winner);
                }
            }

            const totalMatchesCount = matches.reduce((acc, r) => acc + r.length, 0);
            document.getElementById('stat-matches').textContent = totalMatchesCount;

            renderKnockoutUI(totalRounds);
        }

        // Render Knockout Visual UI with Light Fishbone Connectors
        function renderKnockoutUI(totalRounds) {
            const display = document.getElementById('bracket-display');
            display.innerHTML = '';

            matches.forEach((roundMatches, roundIdx) => {
                const roundNum = roundIdx + 1;
                const distance = totalRounds - roundNum;
                let rName = "";

                if (distance === 0) {
                    rName = "รอบชิงชนะเลิศ";
                } else if (distance === 1) {
                    rName = "รอบรองชนะเลิศ";
                } else if (distance === 2) {
                    rName = "รอบก่อนรองชนะเลิศ";
                } else if (roundNum === 1) {
                    rName = "รอบแรก";
                } else {
                    const teamsInRound = Math.pow(2, distance + 1);
                    rName = `รอบ ${teamsInRound} รายชื่อ`;
                }

                const col = document.createElement('div');
                col.className = "flex flex-col justify-around min-w-[300px] sm:min-w-[340px] space-y-6";

                const roundHeader = document.createElement('div');
                roundHeader.className = "text-center mb-2 flex-shrink-0";
                roundHeader.innerHTML = `
                    <h4 class="text-xs font-bold uppercase tracking-wider text-red-600 bg-red-50 py-1.5 px-3 rounded-xl border border-red-100">
                        ${rName}
                    </h4>
                `;
                col.appendChild(roundHeader);

                const matchesDiv = document.createElement('div');
                matchesDiv.className = "flex flex-col justify-around flex-1 space-y-6";

                roundMatches.forEach((m, mIdx) => {
                    const matchBox = document.createElement('div');
                    const isFinal = (roundNum === totalRounds);

                    matchBox.className = `bg-white border ${isFinal ? 'border-red-500 shadow-md ring-2 ring-red-100' : 'border-slate-200'} rounded-2xl p-3.5 shadow-sm flex flex-col justify-center min-h-[110px] transition`;

                    const t1IsBye = m.team1.name === "BYE";
                    const t2IsBye = m.team2.name === "BYE";

                    const t1IsWinner = m.winner && m.winner.name === m.team1.name;
                    const t2IsWinner = m.winner && m.winner.name === m.team2.name;

                    const t1Label = getDisplayLabel(m.team1, m.team1.num);
                    const t2Label = getDisplayLabel(m.team2, m.team2.num);

        // Directly set Win / Lose
        function setWinner(matchId, teamSlot) {
            let match = null;
            let rIdx = -1;

            matches.forEach((round, r) => {
                round.forEach(m => {
                    if (m.id === matchId) {
                        match = m;
                        rIdx = r;
                    }
                });
            });

            if (!match) return;

            const selectedTeam = teamSlot === 1 ? match.team1 : match.team2;
            if (selectedTeam.name === "BYE" || selectedTeam.name.includes("ผู้ชนะ R")) return;

            match.winner = selectedTeam;
            propagateWinner(rIdx, matchId, selectedTeam);
            renderKnockoutUI(matches.length);
        }

        // Forward winner to next match slot
        function propagateWinner(currentRoundIdx, matchId, winnerObj) {
            if (currentRoundIdx + 1 >= matches.length) return;

            const currentRoundMatches = matches[currentRoundIdx];
            const matchIndex = currentRoundMatches.findIndex(m => m.id === matchId);
            const nextRoundMatches = matches[currentRoundIdx + 1];

            const nextMatchIdx = Math.floor(matchIndex / 2);
            const targetMatch = nextRoundMatches[nextMatchIdx];
            const isTeam1 = (matchIndex % 2 === 0);

            if (targetMatch) {
                if (isTeam1) {
                    targetMatch.team1 = winnerObj;
                } else {
                    targetMatch.team2 = winnerObj;
                }
                targetMatch.winner = null;
            }
        }

        // Check and display champion
        function checkChampion(totalRounds) {
            const finalMatch = matches[totalRounds - 1] ? matches[totalRounds - 1][0] : null;
            if (finalMatch && finalMatch.winner && !finalMatch.winner.name.includes("ผู้ชนะ R")) {
                document.getElementById('champion-banner').classList.remove('hidden');
                document.getElementById('champion-name').textContent = getDisplayLabel(finalMatch.winner);
            } else {
                document.getElementById('champion-banner').classList.add('hidden');
            }
        }

        // Round Robin Format Builder
        function buildRoundRobinStandings() {
            let pairs = [];
            for (let i = 0; i < teams.length; i++) {
                for (let j = i + 1; j < teams.length; j++) {
                    const swap = Math.random() < 0.5;
                    const t1 = swap ? j : i;
                    const t2 = swap ? i : j;
                    pairs.push({
                        id: `rr-${i}-${j}`,
                        t1: t1,
                        t2: t2,
                        winner: null
                    });
                }
            }
            shuffleArray(pairs);
            rrMatches = pairs;

            renderRoundRobinUI();
        }

        // Render Round Robin UI (Standings Table + Pairings Cards)

                const p = getParticipantObj(team);
                return {
                    idx: idx + 1,
                    name: p.name,
                    affiliation: p.affiliation,
                    played: 0,
                    won: 0,
                    lost: 0

                if (m.winner !== null) {
                    stats[m.t1].played++;
                    stats[m.t2].played++;

                        stats[m.t2].lost++;

                        stats[m.t2].won++;
                        stats[m.t1].lost++;

                        <td class="p-3 font-bold text-slate-400">${s.idx}</td>
                        <td class="p-3 font-medium text-slate-900">${escapeHtml(s.name)}</td>

                        <h3 class="font-bold text-slate-900 text-sm">ตารางจับคู่การแข่งขัน (สุ่มจับคู่ตามลำดับ)</h3>
                        <button onclick="shuffleRRMatches()" class="text-xs bg-slate-100 hover:bg-slate-200 text-slate-700 px-3 py-1.5 rounded-xl font-medium transition">

'bg-emerald-50 border-emerald-500 text-emerald-900 font-bold' : (t1IsWinner ? 'bg-slate-50 border-slate-200 text-slate-400 opacity-60' : 'bg-slate-50 border-slate-200 text-slate-700 hover:border-slate-300')}">

                    </div>
                `;

                    </div>
                </div>
            `;

        // Set Round Robin Winner
        function setRRWinner(matchId, winnerIdx) {
            const match = rrMatches.find(m => m.id === matchId);
            if (match) {
                match.winner = (match.winner === winnerIdx) ? null : winnerIdx;
                renderRoundRobinUI();
            }
        }

        // Shuffle Round Robin Match Order
        function shuffleRRMatches() {
            shuffleArray(rrMatches);
            renderRoundRobinUI();
        }

        // Custom Modal Alert Replacement
        function alertModal(msg) {
            const modal = document.createElement('div');
            modal.className = "fixed inset-0 bg-slate-900/40 backdrop-blur-sm z-50 flex items-center justify-center p-4";
            modal.innerHTML = `
                <div class="bg-white border border-slate-200 rounded-2xl max-w-xs w-full p-5 text-center shadow-xl">
                    <p class="text-xs text-slate-800 font-medium mb-4">${escapeHtml(msg)}</p>
                    <button onclick="this.closest('.fixed').remove()" class="w-full py-2 bg-red-600 hover:bg-red-700 text-white font-bold text-xs rounded-xl transition">
                        ตกลง
                    </button>
                </div>
            `;
            document.body.appendChild(modal);
        }

        // Escape HTML
        function escapeHtml(str) {
            return String(str)
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }
